add_library(SoundControlWindowsShared STATIC Shared.cpp)
target_link_libraries(SoundControlWindowsShared PUBLIC SoundControlCore ole32 shell32 uuid)
target_include_directories(SoundControlWindowsShared PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(SoundControlWindowsShared PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)

add_library(SoundControlAPO SHARED Apo.cpp Apo.def)
target_link_libraries(SoundControlAPO PRIVATE SoundControlWindowsShared)
set_target_properties(SoundControlAPO PROPERTIES PREFIX "")

add_executable(SoundControlWindows WIN32 App.cpp)
target_link_libraries(SoundControlWindows PRIVATE SoundControlWindowsShared comctl32 comdlg32 gdiplus dwmapi)
if(MINGW)
    target_link_options(SoundControlWindows PRIVATE -municode)
    target_link_options(SoundControlAPO PRIVATE -static-libgcc -static-libstdc++ -static)
    target_link_options(SoundControlWindows PRIVATE -static-libgcc -static-libstdc++ -static)
endif()
if(MSVC)
    target_compile_options(SoundControlAPO PRIVATE /W4 /permissive- /EHsc)
    target_compile_options(SoundControlWindows PRIVATE /W4 /permissive- /utf-8 /EHsc)
endif()
install(TARGETS SoundControlWindows SoundControlAPO RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES "${PROJECT_SOURCE_DIR}/Scripts/install-windows.ps1" "${PROJECT_SOURCE_DIR}/Scripts/uninstall-windows.ps1" "${PROJECT_SOURCE_DIR}/Scripts/WindowsInstall.psm1" DESTINATION .)

add_executable(SoundControlApoTests "${PROJECT_SOURCE_DIR}/Tests/WindowsApoTests.cpp" Shared.cpp)
target_include_directories(SoundControlApoTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(SoundControlApoTests PRIVATE SOUNDCONTROL_TESTING UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(SoundControlApoTests PRIVATE SoundControlCore ole32 shell32 uuid)
if(MINGW)
    target_link_options(SoundControlApoTests PRIVATE -static)
endif()
add_test(NAME SoundControlApoTests COMMAND SoundControlApoTests)

add_dependencies(SoundControlApoTests SoundControlAPO)

install(FILES "${PROJECT_SOURCE_DIR}/docs/windows.md" "${PROJECT_SOURCE_DIR}/docs/windows-validation.md" DESTINATION .)
