add_library(MacToolsWindowsShared STATIC Shared.cpp)
target_link_libraries(MacToolsWindowsShared PUBLIC MacToolsCore ole32 shell32 uuid)
target_include_directories(MacToolsWindowsShared PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(MacToolsWindowsShared PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)

add_library(MacToolsAPO SHARED Apo.cpp Apo.def)
target_link_libraries(MacToolsAPO PRIVATE MacToolsWindowsShared)
set_target_properties(MacToolsAPO PROPERTIES PREFIX "")

add_executable(MacToolsWindows WIN32 App.cpp)
target_link_libraries(MacToolsWindows PRIVATE MacToolsWindowsShared comctl32 comdlg32 gdiplus dwmapi)
if(MINGW)
    target_link_options(MacToolsWindows PRIVATE -municode)
    target_link_options(MacToolsAPO PRIVATE -static-libgcc -static-libstdc++ -static)
    target_link_options(MacToolsWindows PRIVATE -static-libgcc -static-libstdc++ -static)
endif()
if(MSVC)
    target_compile_options(MacToolsAPO PRIVATE /W4 /permissive- /EHsc)
    target_compile_options(MacToolsWindows PRIVATE /W4 /permissive- /utf-8 /EHsc)
endif()
install(TARGETS MacToolsWindows MacToolsAPO RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES "${PROJECT_SOURCE_DIR}/Scripts/install-windows.ps1" "${PROJECT_SOURCE_DIR}/Scripts/uninstall-windows.ps1" "${PROJECT_SOURCE_DIR}/Scripts/WindowsInstall.psm1" DESTINATION .)

add_executable(MacToolsApoTests "${PROJECT_SOURCE_DIR}/Tests/WindowsApoTests.cpp" Shared.cpp)
target_include_directories(MacToolsApoTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(MacToolsApoTests PRIVATE MACTOOLS_TESTING UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(MacToolsApoTests PRIVATE MacToolsCore ole32 shell32 uuid)

    target_link_options(MacToolsApoTests PRIVATE -static)
endif()
add_test(NAME MacToolsApoTests COMMAND MacToolsApoTests)

add_dependencies(MacToolsApoTests MacToolsAPO)
